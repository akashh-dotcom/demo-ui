##
# Default server configuration
##

server {

    server_name zentrovia.tech  epubconverter.zentrovia.tech;
	client_max_body_size 200M; # increase this if you need to increase the upload size
	
    # =========================
    # FRONTEND (STATIC REACT BUILD)
    # =========================
    root /var/www/html;
    index index.html index.htm;

    location / {
        # React SPA routing
        try_files $uri $uri/ /index.html;
    }

    # =========================
    # BACKEND API (Node.js)
    # =========================
    location /api/ {
        proxy_pass http://localhost:5000;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # =========================
    # PDF API
    # =========================
    location /pdf/ {
        proxy_pass http://localhost:8000;
        proxy_http_version 1.1;
    }

    # =========================
    # EPUB API
    # =========================
    location /epub/ {
        proxy_pass http://localhost:5001;
        proxy_http_version 1.1;
    }

    # =========================
    # EPUB EDITOR UI
    # =========================
    location /epub-editor/ {
        proxy_pass http://localhost:5002;
        proxy_http_version 1.1;
    }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/epubconverter.zentrovia.tech/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/epubconverter.zentrovia.tech/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}

server {
    if ($host = epubconverter.zentrovia.tech) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80 default_server;
    listen [::]:80 default_server;

    server_name darkclouds.online www.darkclouds.online epubconverter.zentrovia.tech;
    return 404; # managed by Certbot

}
